'use client';

import { useState, useTransition } from 'react';
import { Lock, Plus, Trash2 } from 'lucide-react';
import type { Category } from '@/lib/types';
import { addCategory, removeCategory } from './actions';

const COLORS = ['#f59e0b', '#10b981', '#3b82f6', '#ef4444', '#a855f7', '#64748b'];

export default function CategoriesSection({
  isPro,
  categories,
}: {
  isPro: boolean;
  categories: Category[];
}) {
  const [name, setName] = useState('');
  const [color, setColor] = useState(COLORS[0]);
  const [error, setError] = useState<string | null>(null);
  const [, startTransition] = useTransition();

  function add() {
    setError(null);
    if (!name.trim()) return;
    addCategory(name.trim(), color).then((res) => {
      if (!res.ok) setError(res.error);
      else setName('');
    });
  }

  return (
    <section className="rounded-2xl border border-stone-200 bg-white p-6">
      <div className="flex items-center justify-between">
        <div>
          <h2 className="text-lg font-semibold">Categories</h2>
          <p className="mt-1 text-sm text-stone-600">
            Tag documents with categories like "Passport" or "Insurance".
          </p>
        </div>
        {!isPro && (
          <span className="inline-flex items-center gap-1 rounded-full bg-stone-100 px-2.5 py-1 text-xs font-medium text-stone-600">
            <Lock className="h-3 w-3" /> Pro
          </span>
        )}
      </div>

      {isPro ? (
        <>
          <div className="mt-4 flex flex-wrap items-center gap-2">
            <input
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder="New category name"
              className="rounded-lg border border-stone-300 px-3 py-2 text-sm focus:border-brand-500 focus:outline-none focus:ring-2 focus:ring-brand-200"
            />
            <div className="flex items-center gap-1">
              {COLORS.map((c) => (
                <button
                  key={c}
                  type="button"
                  onClick={() => setColor(c)}
                  className={
                    'h-7 w-7 rounded-full border-2 ' +
                    (color === c ? 'border-stone-900' : 'border-transparent')
                  }
                  style={{ backgroundColor: c }}
                  aria-label={`Pick color ${c}`}
                />
              ))}
            </div>
            <button
              onClick={add}
              className="inline-flex items-center gap-1.5 rounded-lg bg-stone-900 px-3 py-2 text-sm font-semibold text-white hover:bg-stone-800"
            >
              <Plus className="h-4 w-4" /> Add
            </button>
          </div>
          {error && (
            <p className="mt-2 rounded-md bg-red-50 px-3 py-2 text-sm text-red-700">{error}</p>
          )}

          <ul className="mt-4 divide-y divide-stone-100">
            {categories.length === 0 && (
              <li className="py-3 text-sm text-stone-500">No categories yet.</li>
            )}
            {categories.map((c) => (
              <li key={c.id} className="flex items-center justify-between py-2.5">
                <div className="flex items-center gap-2">
                  <span
                    className="h-3 w-3 rounded-full"
                    style={{ backgroundColor: c.color }}
                  />
                  <span className="text-sm font-medium">{c.name}</span>
                </div>
                <button
                  onClick={() => {
                    if (!confirm(`Delete category "${c.name}"?`)) return;
                    startTransition(() => { removeCategory(c.id); });
                  }}
                  className="rounded-md p-1.5 text-stone-500 hover:bg-red-50 hover:text-red-700"
                  aria-label="Delete category"
                >
                  <Trash2 className="h-4 w-4" />
                </button>
              </li>
            ))}
          </ul>
        </>
      ) : (
        <p className="mt-3 text-sm text-stone-500">
          Categories are a Pro feature. Upgrade in the Billing section above.
        </p>
      )}
    </section>
  );
}
